	我们知道椭圆具有这样的光学性质:
	由焦点$ F_1$ 发出的一条光线$\bvec{F_1P}$,经椭圆面反射,出射光线经过
另一焦点	$ F_2$

即 
入射光线为$\bvec{F_1P}$,出射光线为$\bvec{PF_2}$




光学性质有多样的证明方法,下面给出一种较为直接的证法





欲证光学性质,只需证$P$点的法线平分$\angle F_1PF_2$

只需证$P$点的法向量$\vec{n}$与$\angle F_1PF_2$的角平分线共线





设椭圆$C$焦点$F_1 \left( -c,0\right)$ \quad  $F_2 \left( c,0\right)$
\ ,  \qquad 
长轴长为$2a$

椭圆上任意一点$P$$\left( x_0,y_0\right)$ 
\subsection{求$\angle F_1PF_2$的角平分线}
\begin{figure}[h]
	\centering
	\includegraphics[height=6cm]{角平分线1}
	\caption{}
	\label{fig:}
\end{figure}

设$\vec{u}$ 为$PF_1$的单位向量,$\vec{v}$ 为$PF_2$的单位向量

 \begin{equation*} 
 \begin{split}
 	& \vec{u}=\frac{\bvec{PF_1}}{\left| PF_1 \right| }
 		\qquad
      \vec{v}=\frac{\bvec{PF_2}}{\left| PF_2 \right| }\\
 	& \vec{w}=\vec{u}+\vec{v}
 \end{split}
\end{equation*}
则$PADB$为菱形,$PD$为$PA$与$PB$的角平分线,
即$\vec{w}$与$\angle F_1PF_2$的角平分线共线
\begin{equation*}
\vec{w}=
\frac{\bvec{PF_1}}{\left| PF_1 \right| }+\frac{\bvec{PF_2}}{\left| PF_2 \right| }
\end{equation*}

\subsection{求$P$点法向量$\vec{n}$}
先写出椭圆方程
\begin{equation*}	
\sqrt{\left( x+c \right) ^2+y ^2}+
\sqrt{\left( x-c \right) ^2+y ^2}
=2a
\end{equation*}
\subsubsection{法一}
\begin{subequations}
	\begin{align*}
		&\text{两侧对$x$求导}\\
		&\frac{x+c}{\sqrt{\left( x+c \right) ^2+y ^2}}
		+\frac{y}{\sqrt{\left( x+c \right) ^2+y ^2}} \dfrac{dy}{dx}+
		\frac{x-c}{\sqrt{\left( x-c \right) ^2+y^2}}
		+\frac{y}{\sqrt{\left( x-c \right) ^2+y ^2}}\dfrac{dy}{dx}=
		0\\
		&\text{可以解得} \frac{dy}{dx}\\
		&P \text{点切线斜率为} \left. \frac{dy}{dx} \right| _{x=x_0}
		\ , \  \text{切线方向向量$\vec{l}= \left( 1,\left. \frac{dy}{dx} \right|  _{x=x_0} \right)$ }
	\end{align*}
\end{subequations}
	则
\begin{equation*}
\begin{split}
	\vec{n}& =  \left( \left. \frac{dy}{dx} \right|  _{x=x_0},-1 \right)\\
	 &=   \left( \frac{x+c}{\sqrt{\left( x+c \right) ^2+y ^2}}
		+\frac{x-c}{\sqrt{\left( x-c \right) ^2+y^2}} , 
		\frac{y}{\sqrt{\left( x+c \right) ^2+y ^2}}
		+\frac{y}{\sqrt{\left( x-c \right) ^2+y ^2}} \right) \\
	&=\frac{\bvec{F_1P}}{\left|PF_1\right|}+\frac{\bvec{F_2P}}{\left| PF_2 \right| }
\end{split}
\end{equation*}
\subsubsection{法二}
设$$ f \left( x,y\right) =\sqrt{\left( x+c \right) ^2+y ^2}+
\sqrt{\left( x-c \right) ^2+y ^2}$$
 则椭圆$C$是$f$的一条等值线
 
 由$\nabla f $与$f \left( x,y\right) $等值线垂直
 
 得\begin{equation*}
 	\begin{split}
 		\vec{n}=&\nabla f\\
        =& \left( f_x^{'},f_y^' \right)\\
        =&    \left( \frac{x+c}{\sqrt{\left( x+c \right) ^2+y ^2}}
        +\frac{x-c}{\sqrt{\left( x-c \right) ^2+y^2}} , 
        \frac{y}{\sqrt{\left( x+c \right) ^2+y ^2}}
        +\frac{y}{\sqrt{\left( x-c \right) ^2+y ^2}} \right) \\
        = & 	\frac{\bvec{F_1P}}{\left|PF_1\right|}+\frac{\bvec{F_2P}}{\left| PF_2 \right| }
 	\end{split}
 \end{equation*}
\subsection{结论}
综上
\begin{equation*}
\begin{split}
		&\vec{w}=
	\frac{\bvec{PF_1}}{\left| PF_1 \right| }+\frac{\bvec{PF_2}}{\left| PF_2 \right| }\\
	&\vec{n}=\frac{\bvec{F_1P}}{\left|PF_1\right|}+\frac{\bvec{F_2P}}{\left| PF_2 \right| }
\end{split}
 \end{equation*}
\qquad  \qquad 故$$  \vec{w} \textfractionsolidus\textfractionsolidus  \vec{n}$$
\centering 光学性质得证
